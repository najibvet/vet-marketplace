async function submitBooking(e) { e.preventDefault(); const submitBtn = document.getElementById('submit-btn'); submitBtn.innerText = "Mengunggah & Menyimpan..."; submitBtn.disabled = true; const serviceId = document.getElementById('selected-service-id').value; const serviceDetail = document.getElementById('selected-service-name').value; const amount = Number(document.getElementById('selected-service-price').value); const clientName = document.getElementById('client-name').value; const clientPhone = document.getElementById('client-phone').value; const petInfo = document.getElementById('pet-info').value; const proofFile = document.getElementById('booking-proof').files[0]; const platformFee = Math.round(amount * 0.15); let proofUrl = null; try { // 1. Jika ada file bukti transfer, upload ke Supabase Storage if (proofFile) { const fileExt = proofFile.name.split('.').pop(); const fileName = `${Date.now()}_${Math.random().toString(36).substring(2, 7)}.${fileExt}`; const { error: uploadError } = await supabaseClient.storage .from('proofs') .upload(fileName, proofFile); if (!uploadError) { const { data: urlData } = supabaseClient.storage .from('proofs') .getPublicUrl(fileName); proofUrl = urlData.publicUrl; } } // 2. Simpan ke database Supabase await supabaseClient.from('transactions').insert([ { service_id: serviceId, service_detail: serviceDetail, total_amount: amount, platform_fee: platformFee, client_name: clientName, client_phone: clientPhone, pet_info: petInfo, payment_status: 'PENDING', status: 'pending', proof_url: proofUrl } ]); } catch (err) { console.error("Error saving transaction:", err); } alert('Pesanan berhasil dibuat! Anda akan diarahkan ke WhatsApp.'); // 3. Format pesan WhatsApp agar menyertakan link bukti transfer const adminPhone = '6281215667746'; // Nomor WhatsApp Admin let waText = `Halo Admin VetBook, saya ingin konfirmasi pembayaran booking layanan:\n\n*Layanan:* ${serviceDetail}\n*Nama:* ${clientName} (${clientPhone})\n*Pasien:* ${petInfo}\n*Total Biaya:* Rp ${amount.toLocaleString('id-ID')}\n`; if (proofUrl) { waText += `\n*📷 Link Bukti Transfer:* ${proofUrl}\n`; } waText += `\nMohon verifikasi pembayarannya. Terima kasih.`; // Arahkan ke WhatsApp Admin dengan membawa link bukti transfer window.location.href = `https://wa.me/${adminPhone}?text=${encodeURIComponent(waText)}`; }
